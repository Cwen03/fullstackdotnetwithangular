## Lab 2: Component with Input Property

**Objective**: Pass data to a component using @Input.

**Instructions**:
1. Generate a component named `greeting`.
2. Pass a name as an @Input and display "Hello, {{name}}!".

**Solution**:
```bash
ng generate component greeting
```
```typescript
// greeting.component.ts
import { Component, Input } from '@angular/core';

@Component({
  selector: 'app-greeting',
  templateUrl: './greeting.component.html'
})
export class GreetingComponent {
  @Input() name: string = '';
}
```
```html
<!-- greeting.component.html -->
<p>Hello, {{name}}!</p>
```
```html
<!-- app.component.html -->
<app-greeting [name]="'Alice'"></app-greeting>
```

## Lab 3: Component with Output Event

**Objective**: Emit an event from a child component.

**Instructions**:
1. Generate a component named `button-clicker`.
2. Emit an event when a button is clicked and handle it in the parent.

**Solution**:
```bash
ng generate component button-clicker
```
```typescript
// button-clicker.component.ts
import { Component, EventEmitter, Output } from '@angular/core';

@Component({
  selector: 'app-button-clicker',
  templateUrl: './button-clicker.component.html'
})
export class ButtonClickerComponent {
  @Output() clicked = new EventEmitter<string>();

  handleClick() {
    this.clicked.emit('Button was clicked!');
  }
}
```
```html
<!-- button-clicker.component.html -->
<button (click)="handleClick()">Click Me</button>
```
```html
<!-- app.component.html -->
<app-button-clicker (clicked)="handleEvent($event)"></app-button-clicker>
<p>{{ message }}</p>
```
```typescript
// app.component.ts
export class AppComponent {
  message = '';

  handleEvent(data: string) {
    this.message = data;
  }
}
```

## Lab 9: Component Communication via Service

**Objective**: Share data between two components using a service.

**Instructions**:
1. Create `sender` and `receiver` components.
2. Use a service to share data between them.

**Solution**:
```bash
ng generate component sender
ng generate component receiver
ng generate service message
```
```typescript
// message.service.ts
import { Injectable } from '@angular/core';
import { BehaviorSubject } from 'rxjs';

@Injectable({
  providedIn: 'root'
})
export class MessageService {
  private messageSource = new BehaviorSubject<string>('');
  currentMessage = this.messageSource.asObservable();

  changeMessage(message: string) {
    this.messageSource.next(message);
  }
}
```
```typescript
// sender.component.ts
import { MessageService } from '../message.service';

export class SenderComponent {
  constructor(private messageService: MessageService) {}

  sendMessage() {
    this.messageService.changeMessage('Hello from Sender!');
  }
}
```
```html
<!-- sender.component.html -->
<button (click)="sendMessage()">Send Message</button>
```
```typescript
// receiver.component.ts
import { MessageService } from '../message.service';

export class ReceiverComponent {
  message = '';

  constructor(private messageService: MessageService) {
    this.messageService.currentMessage.subscribe(msg => this.message = msg);
  }
}
```
```html
<!-- receiver.component.html -->
<p>Received: {{ message }}</p>
```

## Lab 10: Dynamic Component Loader with ViewContainerRef

**Objective**: Dynamically load a component at runtime.

**Instructions**:
1. Dynamically load the `hello-world` component inside another component using `ViewContainerRef`.

**Solution**:
```typescript
// app.component.ts
import { Component, ViewChild, ViewContainerRef, ComponentRef } from '@angular/core';
import { HelloWorldComponent } from './hello-world/hello-world.component';

@Component({
  selector: 'app-root',
  templateUrl: './app.component.html'
})
export class AppComponent {
  @ViewChild('container', { read: ViewContainerRef }) container!: ViewContainerRef;
  componentRef!: ComponentRef<HelloWorldComponent>;

  loadComponent() {
    this.container.clear();
    this.componentRef = this.container.createComponent(HelloWorldComponent);
  }
}
```
```html
<!-- app.component.html -->
<button (click)="loadComponent()">Load HelloWorld Component</button>
<ng-template #container></ng-template>
```
