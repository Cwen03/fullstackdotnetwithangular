### **Lab 8: Custom Pipe (Reverse String)**

ng generate pipe reverse

**Task:** Create a custom pipe to reverse a string.

```typescript
// reverse.pipe.ts
import { Pipe, PipeTransform } from '@angular/core';

@Pipe({ name: 'reverse' })
export class ReversePipe implements PipeTransform {
  transform(value: string): string {
    return value.split('').reverse().join('');
  }
}
```

```html
<p>Original: {{ message }}</p>
<p>Reversed: {{ message | reverse }}</p>
```

```typescript
export class AppComponent {
  message: string = 'Custom Pipe Example';
}
```

### **Lab 9: Custom Pipe with Parameters (Filter Array)**

**Task:** Create a custom pipe to filter an array based on a search term.

```typescript
// filter.pipe.ts
import { Pipe, PipeTransform } from '@angular/core';

@Pipe({ name: 'filter' })
export class FilterPipe implements PipeTransform {
  transform(items: string[], search: string): string[] {
    if (!search) return items;
    return items.filter(item => item.toLowerCase().includes(search.toLowerCase()));
  }
}
```

```html
<input [(ngModel)]="searchTerm" placeholder="Search item" />
<ul>
  <li *ngFor="let item of items | filter:searchTerm">{{ item }}</li>
</ul>
```

```typescript
export class AppComponent {
  items: string[] = ['Angular', 'React', 'Vue', 'Svelte', 'Next'];
  searchTerm: string = '';
}
```

### **Lab 10: Async Pipe**

**Task:** Use async pipe to display data from an observable.

```typescript
// app.component.ts
import { Component } from '@angular/core';
import { of, delay } from 'rxjs';

@Component({
  selector: 'app-root',
  templateUrl: './app.component.html'
})
export class AppComponent {
  data$ = of('Async Pipe Example').pipe(delay(2000));
}
```

```html
<p>Data: {{ data$ | async }}</p>
```
